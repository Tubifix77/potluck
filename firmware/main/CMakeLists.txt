# M6.1: the extender variant pulls in poor-mans-extender's pme_hotspot. The directory arrives through
# the environment (tools\build_firmware.ps1 -PmeHotspot), which -- unlike a Kconfig value -- is
# visible while ESP-IDF expands component requirements.
set(pot_extra_requires "")
if(DEFINED ENV{POT_PME_HOTSPOT_DIR})
  list(APPEND pot_extra_requires pme_hotspot)
endif()

# M8.2 (PS-1): every component under the -AppComponents directories that carries a
# potluck-modules.json is an external actor component. It defines, in namespace pot,
#   const ActorKind* app_rows_<directory name>(size_t* n);
# and this file generates the list of them, so no Potluck source names one.
set(pot_app_components "")
if(DEFINED ENV{POT_APP_COMPONENT_DIRS})
  foreach(d $ENV{POT_APP_COMPONENT_DIRS})
    file(GLOB children LIST_DIRECTORIES true "${d}/*")
    foreach(c ${children})
      if(IS_DIRECTORY "${c}" AND EXISTS "${c}/potluck-modules.json" AND EXISTS "${c}/CMakeLists.txt")
        get_filename_component(cname "${c}" NAME)
        list(APPEND pot_app_components "${cname}")
      endif()
    endforeach()
  endforeach()
endif()
list(APPEND pot_extra_requires ${pot_app_components})
set(pot_app_rows_src "${CMAKE_CURRENT_BINARY_DIR}/app_rows.cpp")
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
  set(decls "")
  set(entries "")
  foreach(c ${pot_app_components})
    string(APPEND decls "const ActorKind* app_rows_${c}(size_t* n);\n")
    string(APPEND entries "    {\"${c}\", &app_rows_${c}},\n")
  endforeach()
  list(LENGTH pot_app_components napps)
  file(WRITE "${pot_app_rows_src}.tmp"
    "// Generated by firmware/main/CMakeLists.txt from -AppComponents (M8.2, PS-1). Do not edit.\n"
    "#include \"actor_table.hpp\"\n"
    "namespace pot {\n${decls}namespace app {\n"
    "const AppRows kAppRows[] = {\n${entries}    {nullptr, nullptr},\n};\n"
    "const size_t kAppRowsLen = ${napps};\n"
    "}  // namespace app\n}  // namespace pot\n")
  # Only touch the real file when it changes, so an unchanged list does not rebuild main.
  file(COPY_FILE "${pot_app_rows_src}.tmp" "${pot_app_rows_src}" ONLY_IF_DIFFERENT)
endif()

idf_component_register(
  SRCS "m0_main.cpp" "selftest.cpp" "crypto_bench.cpp" "extender.cpp" "actor_table.cpp" "${pot_app_rows_src}"
  INCLUDE_DIRS "."
  # esp_driver_gpio, not the old monolithic `driver`: ESP-IDF v6.0 split the drivers into
  # per-peripheral components and `driver` no longer re-exports driver/gpio.h.
  REQUIRES pot_frame pot_link pot_espnow pot_deploy pot_can esp_timer esp_driver_gpio esp_driver_rmt esp_app_format monocypher mbedtls pot_trust pot_reconcile pot_actors esp_driver_uart esp_driver_tsens ${pot_extra_requires}
)
